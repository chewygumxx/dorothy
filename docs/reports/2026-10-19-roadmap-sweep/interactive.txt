const root = '<probe-root>';
const [mode, name, ...args] = process.argv.slice(2);
const log = (value: unknown) => appendFileSync(`${root}/${name}.observations.jsonl`, JSON.stringify(value) + '\n');
const base = new URL(process.env.ANTHROPIC_BASE_URL || 'https://api.anthropic.com');
const inspector = Bun.serve({hostname: '127.0.0.1', port: 0, idleTimeout: 0,
    async fetch(request) {
        const url = new URL(request.url);
        const bytes = request.method === 'GET' || request.method === 'HEAD' ? undefined : await request.arrayBuffer();
        if (bytes !== undefined && /\/v1\/messages(?:\/count_tokens)?$/.test(url.pathname)) {
            try {
                const body = JSON.parse(new TextDecoder().decode(bytes));
                const text = JSON.stringify({system: body.system, messages: body.messages});
                const systems = typeof body.system === 'string' ? [body.system] : (body.system ?? []).map((b: any) => b.text ?? '');
                log({kind:'request', path:url.pathname, model:body.model,
                    systemBlocks: systems.length,
                    identity:systems.some((s: string) => s.includes('You are Dorothy,')),
                    markerLeft:systems.some((s: string) => s.includes(MARKER)),
                    rootGuide:text.includes('# AGENTS.md'),
                    packageGuide:text.includes('# packages/tui/AGENTS.md'),
                    memoryIndex:text.includes('./memory-priorities.md') && text.includes('./target-harnesses.md'),
                    memoryHook:text.includes('Memory ranks compatibility, access, progressive disclosure, persistence')});
            } catch { log({kind:'unparsed-request', path:url.pathname}); }
        }
        const headers = new Headers(request.headers);
        headers.delete('host'); headers.delete('connection');
        const target = new URL(url.pathname + url.search, base);
        const reply = await fetch(target, {method:request.method, headers, body:bytes, signal:request.signal, redirect:'manual'});
        const responseHeaders = new Headers(reply.headers);
        responseHeaders.delete('content-encoding'); responseHeaders.delete('content-length'); responseHeaders.delete('transfer-encoding');
        return new Response(reply.body, {status:reply.status, headers:responseHeaders});
    }
});

probe.py (launch, environment and result helper):
import os, json, re, subprocess, sys, uuid
from pathlib import Path
ROOT=Path('<probe-root>')
REPO=Path('<checkout>')
BUN='/home/chewygumxx/.local/share/mise/installs/bun/1.4.2/bin/bun'

def env_for(name):
    # Match the instructed pre-probe check, printing names rather than values.
    check=subprocess.run("env | grep CLAUDE_CODE_ | sed 's/=.*/=<removed for probe>/'",shell=True,capture_output=True,text=True)
    print('Parent CLAUDE_CODE_ check: '+(check.stdout.strip() or 'none'), flush=True)
    env={k:v for k,v in os.environ.items() if not k.startswith('CLAUDE_CODE_') and not k.startswith('npm_') and k!='CLAUDECODE'}
    for key,part in [('XDG_DATA_HOME','data'),('XDG_STATE_HOME','state'),('XDG_CACHE_HOME','cache'),('XDG_CONFIG_HOME','config'),('XDG_RUNTIME_DIR','runtime')]:
        env[key]=str(ROOT/name/part)
    env['CLAUDE_CONFIG_DIR']=os.environ.get('CLAUDE_CONFIG_DIR',str(Path.home()/'.claude'))
    env['PATH']=str(Path(BUN).parent)+':/opt/claude-code/bin:/home/chewygumxx/.local/share/mise/installs/node/24/bin:'+env['PATH']
    env.pop('FORCE_COLOR',None)
    env['ANTHROPIC_MODEL']='haiku'
    return env

def scrub(text):
    text=re.sub(r'\x1b\[[0-?]*[ -/]*[@-~]','',text)
    text=re.sub(r'[A-Za-z0-9_.+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}', '<account-redacted>',text)
    text=re.sub(r'sk-ant-[A-Za-z0-9_-]+','<credential-redacted>',text)
    text=re.sub(r'(?i)(bearer\s+)[^\s"\']+',r'\1<credential-redacted>',text)
    text=re.sub(r'\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b','<session>',text)
    return text.replace('\u2014','-')

if __name__=='__main__':
    mode,name,*args=sys.argv[1:]
    env=env_for(name if name in ['package','memory','proxy','compaction'] else 'interactive')
    if mode=='compaction':
        env['DOTENVX_CONFIG']=os.environ.get('DOTENVX_CONFIG',str(Path(os.environ.get('XDG_CONFIG_HOME',str(Path.home()/'.config')))/'dotenvx'))
        os.chdir(REPO)
        os.execve(BUN,[BUN,'x','--no-install','dotenvx','run','--quiet','--',BUN,'--no-env-file',str(REPO/'packages/cli/src/index.ts')],env)
    if mode=='interactive':
        os.execve(BUN,[BUN,'--no-env-file',str(ROOT/'observe.ts'),'dorothy',name,*args],env)
    cwd=REPO/'packages/tui/src' if name=='package' else REPO if name=='memory' else ROOT/'interactive/cwd'
    command=[BUN,'--no-env-file',str(ROOT/'observe.ts'),mode,name,*args]
    print('Starting probe '+name,flush=True)
    result=subprocess.run(command,cwd=cwd,env=env,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,timeout=150)
    try:
        data=json.loads(result.stdout)
        if isinstance(data,list): data=next(d for d in reversed(data) if d.get('type')=='result')
        summary={k:data.get(k) for k in ['type','subtype','is_error','result','total_cost_usd','num_turns','usage']}
        if data.get('session_id'):
            (ROOT/(name+'.session')).write_text(data['session_id'])
        text=json.dumps(summary,ensure_ascii=False,indent=2)
    except Exception:
        text=result.stdout
    (ROOT/(name+'.result.txt')).write_text(scrub(text)+'\nstderr: '+scrub(result.stderr)+'\nexit: '+str(result.returncode)+'\n')
    print(scrub(text))
    print('stderr: '+scrub(result.stderr))
    print('exit: '+str(result.returncode))
    sys.exit(result.returncode)
