import json,select,socket,socketserver,subprocess,threading,urllib.parse
from pathlib import Path
from probe import ROOT,BUN,env_for,scrub
connections=[]
class Proxy(socketserver.BaseRequestHandler):
    def handle(self):
        data=b''
        while b'\r\n\r\n' not in data and len(data)<65536:
            part=self.request.recv(8192)
            if not part:return
            data+=part
        head,_,tail=data.partition(b'\r\n\r\n')
        method,target,_=head.split(b'\r\n',1)[0].decode().split(' ',2)
        if method=='CONNECT':
            host,port=target.rsplit(':',1);port=int(port)
        else:
            url=urllib.parse.urlsplit(target);host=url.hostname;port=url.port or 80
        allowed=host in ['127.0.0.1','localhost','api.anthropic.com']
        connections.append({'method':method,'host':host if allowed else '<unrelated-host>','port':port,'allowed':allowed})
        if not allowed:
            self.request.sendall(b'HTTP/1.1 502 Probe restricts destinations\r\nContent-Length: 0\r\n\r\n');return
        with socket.create_connection((host,port),timeout=20) as upstream:
            upstream.settimeout(None)
            if method=='CONNECT':
                self.request.sendall(b'HTTP/1.1 200 Connection Established\r\n\r\n')
                if tail:upstream.sendall(tail)
            else:
                parts=head.split(b'\r\n');parts[0]=f'{method} {url.path or "/"}{"?"+url.query if url.query else ""} HTTP/1.1'.encode()
                upstream.sendall(b'\r\n'.join(parts)+b'\r\n\r\n'+tail)
            while True:
                readable,_,_=select.select([self.request,upstream],[],[],20)
                if not readable:return
                for source in readable:
                    part=source.recv(65536)
                    if not part:return
                    (upstream if source is self.request else self.request).sendall(part)
class Server(socketserver.ThreadingTCPServer):
    daemon_threads=True
    allow_reuse_address=True
server=Server(('127.0.0.1',0),Proxy)
thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
env=env_for('proxy')
for key in ['HTTP_PROXY','http_proxy','HTTPS_PROXY','https_proxy','ALL_PROXY','all_proxy','NO_PROXY','no_proxy']:
    env.pop(key,None)
env['HTTP_PROXY']=f'http://127.0.0.1:{server.server_address[1]}'
env['NO_PROXY']='';env['no_proxy']=''
try:
    result=subprocess.run([BUN,'--no-env-file',str(ROOT/'observe.ts'),'dorothy','proxy','-p','--model','haiku','--output-format','json','--no-session-persistence','--tools','','--','Probe only: reply with exactly proxy-ok.'],env=env,cwd=ROOT/'proxy/cwd',stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,timeout=100)
    try:
        value=json.loads(result.stdout)
        output={k:value.get(k) for k in ['result','is_error','total_cost_usd','num_turns']}
    except ValueError:output={'stdout':scrub(result.stdout)}
    output.update({'exit':result.returncode,'stderr':scrub(result.stderr),'connections':connections,'proxy_port':server.server_address[1]})
    text=scrub(json.dumps(output,indent=2))
    (ROOT/'proxy.result.txt').write_text(text+'\n');print(text)
finally:
    server.shutdown();server.server_close()
